\newcommand{\ResultGraphic}[3]{%
  \begin{figure}[t]
    \centering
    \IfFileExists{#1}{%
      \includegraphics[width=#2]{#1}%
    }{%
      \fbox{\parbox{0.9\linewidth}{\textbf{شکل پیدا نشد:} \texttt{#1}\\
      فایل را در این مسیر قرار دهید یا نام/مسیر را در متن به‌روزرسانی کنید.}}%
    }%
    \caption{#3}
    \label{fig:#1}
  \end{figure}
}

\newcommand{\ResultGraphicTwo}[6]{%
  \begin{figure}[t]
    \centering
    \begin{subfigure}{0.48\linewidth}
      \centering
      \IfFileExists{#1}{\includegraphics[width=\linewidth]{#1}}{\fbox{\parbox{0.9\linewidth}{پیدا نشد: \texttt{#1}}}}
      \caption{#3}
    \end{subfigure}\hfill
    \begin{subfigure}{0.48\linewidth}
      \centering
      \IfFileExists{#2}{\includegraphics[width=\linewidth]{#2}}{\fbox{\parbox{0.9\linewidth}{پیدا نشد: \texttt{#2}}}}
      \caption{#4}
    \end{subfigure}
    \caption{#5}
    \label{fig:#6}
  \end{figure}
}

\newcommand{\SafeCSVTable}[3]{%
  \begin{table}[t]
    \centering
    \IfFileExists{#1}{%
      \pgfplotstabletypeset[
        col sep=comma,
        string type,
        header=true,
        columns/.style={column type=l},
        every head row/.style={before row=\toprule, after row=\midrule},
        every last row/.style={after row=\bottomrule}
      ]{#1}%
    }{%
      \fbox{\parbox{0.9\linewidth}{\textbf{فایل CSV پیدا نشد:} \texttt{#1}\\
      خروجی معیارها را در این مسیر ذخیره کنید یا مسیر/نام فایل را اصلاح کنید.}}%
    }%
    \caption{#2}
    \label{#3}
  \end{table}
}
