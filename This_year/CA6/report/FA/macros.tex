% -------------------------
% Safe single figure
% Usage:
% \ResultGraphic{path/to/img.png}{0.8\linewidth}{Caption text}{fig:mylabel}
% -------------------------
\newcommand{\ResultGraphic}[4]{%
  \begin{figure}[t]
    \centering
    \IfFileExists{#1}{%
      \includegraphics[width=#2]{#1}%
    }{%
      \fbox{\parbox{0.9\linewidth}{%
        \textbf{شکل پیدا نشد:} \texttt{#1}\\
        فایل را در این مسیر قرار دهید یا نام/مسیر را اصلاح کنید.}}%
    }%
    \caption{#3}
    \label{#4}
  \end{figure}
}

% -------------------------
% Safe two-panel figure
% Usage:
% \ResultGraphicTwo{img1}{img2}{cap1}{cap2}{main caption}{fig:label}
% -------------------------
\newcommand{\ResultGraphicTwo}[6]{%
  \begin{figure}[t]
    \centering
    \begin{subfigure}{0.48\linewidth}
      \centering
      \IfFileExists{#1}{%
        \includegraphics[width=\linewidth]{#1}%
      }{%
        \fbox{\parbox{0.9\linewidth}{پیدا نشد: \texttt{#1}}}%
      }%
      \caption{#3}
    \end{subfigure}\hfill
    \begin{subfigure}{0.48\linewidth}
      \centering
      \IfFileExists{#2}{%
        \includegraphics[width=\linewidth]{#2}%
      }{%
        \fbox{\parbox{0.9\linewidth}{پیدا نشد: \texttt{#2}}}%
      }%
      \caption{#4}
    \end{subfigure}
    \caption{#5}
    \label{#6}
  \end{figure}
}

% -------------------------
% Safe CSV table
% Usage:
% \SafeCSVTable{metrics.csv}{Table caption}{tab:metrics}
% -------------------------
\newcommand{\SafeCSVTable}[3]{%
  \begin{table}[t]
    \centering
    \IfFileExists{#1}{%
      \pgfplotstabletypeset[
        col sep=comma,
        string type,
        header=true,
        columns/.style={column type=l},
        every head row/.style={before row=\toprule, after row=\midrule},
        every last row/.style={after row=\bottomrule}
      ]{#1}%
    }{%
      \fbox{\parbox{0.9\linewidth}{%
        \textbf{فایل CSV پیدا نشد:} \texttt{#1}\\
        مسیر یا نام فایل را اصلاح کنید.}}%
    }%
    \caption{#2}
    \label{#3}
  \end{table}
}

% -------------------------
% Safe code listing
% Usage:
% \SafeListing{file.py}{Caption}{lst:code}
% -------------------------
\newcommand{\SafeListing}[3]{%
  \begin{figure}[t]
    \centering
    \IfFileExists{#1}{%
      \lstinputlisting{#1}%
    }{%
      \fbox{\parbox{0.9\linewidth}{%
        \textbf{فایل کُد پیدا نشد:} \texttt{#1}\\
        مسیر یا نام فایل را اصلاح کنید.}}%
    }%
    \caption{#2}
    \label{#3}
  \end{figure}
}
